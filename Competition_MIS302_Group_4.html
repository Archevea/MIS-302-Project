Conversion Rates

Conversion Rate By Job

conversion_rate <- prop.table(table(bank_data$y))["yes"]

job_df <- bank_data %>%
  group_by(job) %>%
  summarise(
    total = n(),
    yes_count = sum(y == "yes"),
    response_rate = yes_count / total,
    flag = ifelse(response_rate > conversion_rate, 1, 0)
  )

ggplot(job_df, aes(x = reorder(job, response_rate), y = response_rate, fill = factor(flag))) +
  geom_col() +
  coord_flip() +
  geom_hline(yintercept = conversion_rate, linetype = "dashed", color = "red") +
  theme_minimal() +
  labs(title = "Response Rate by Job", y = "Response Rate", x = "", fill = "")
